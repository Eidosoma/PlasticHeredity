% Keep heading customization on LaTeX's native \@startsection path so that
% Pandoc's already-loaded hyperref/bookmark hooks create a distinct, current
% destination for every heading. Needspace runs before the section command,
% preventing an anchor from being left on the preceding page.
\makeatletter
\renewcommand\section{%
  \Needspace{5\baselineskip}%
  \@startsection{section}{1}{\z@}%
    {-2.1em}%
    {0.65em}%
    {\sffamily\bfseries\color{EidosomaInk}\Large}%
}
\renewcommand\subsection{%
  \Needspace{4\baselineskip}%
  \@startsection{subsection}{2}{\z@}%
    {-1.55em}%
    {0.45em}%
    {\sffamily\bfseries\color{EidosomaBlue}\large}%
}
\renewcommand\subsubsection{%
  \Needspace{3\baselineskip}%
  \@startsection{subsubsection}{3}{\z@}%
    {-1.2em}%
    {0.35em}%
    {\sffamily\bfseries\color{EidosomaInk}\normalsize}%
}
\makeatother
